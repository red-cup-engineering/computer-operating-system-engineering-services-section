cmake_minimum_required(VERSION 3.20)
project(windows_host_pressure LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 target required")
endif()
add_library(windows_host_pressure SHARED windows-host-pressure.cpp)
target_compile_features(windows_host_pressure PRIVATE cxx_std_17)
target_link_libraries(windows_host_pressure PRIVATE pdh bcrypt advapi32)
if(MSVC)
  target_compile_options(windows_host_pressure PRIVATE /W4 /WX /EHsc)
endif()

# Finite native ABI assay, invoked once by the hosted Windows consumer.
add_executable(windows_host_pressure_assay windows-host-pressure-assay.cpp)
target_compile_features(windows_host_pressure_assay PRIVATE cxx_std_17)
target_link_libraries(windows_host_pressure_assay PRIVATE windows_host_pressure)
if(MSVC)
  target_compile_options(windows_host_pressure_assay PRIVATE /W4 /WX /EHsc)
endif()
